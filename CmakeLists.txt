cmake_minimum_required(VERSION 3.20)

project(RayMarching LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_subdirectory(external/glfw)

add_executable(RayMarching
    src/main.cpp
    src/glad.c
    external/imgui/imgui.cpp
    external/imgui/imgui_draw.cpp
    external/imgui/imgui_tables.cpp
    external/imgui/imgui_widgets.cpp
    external/imgui/imgui_demo.cpp
    external/imgui/backends/imgui_impl_glfw.cpp
    external/imgui/backends/imgui_impl_opengl3.cpp
)

target_include_directories(RayMarching
    PRIVATE
        include
        external/imgui
        external/imgui/backends
)

target_link_libraries(RayMarching
    PRIVATE
        glfw
        opengl32
)

target_sources(RayMarching
    PRIVATE
        assets/shaders/basic.glsl
        assets/shaders/basic.frag
        assets/shaders/worldlib.glsl
        assets/shaders/raymarch.frag
        assets/shaders/fractal.frag
        assets/shaders/large_fractal.frag
        assets/shaders/clouds.frag
        assets/shaders/efficient_clouds.frag
        assets/shaders/non_euclidean.frag
        assets/shaders/blood_stream.frag
)

# Copy each asset as a real build step with declared inputs/outputs,
# so editing a shader makes the build out of date and re-copies it.
file(GLOB_RECURSE ASSET_FILES RELATIVE "${CMAKE_SOURCE_DIR}" CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/assets/*")

set(ASSET_OUTPUTS "")
foreach(asset IN LISTS ASSET_FILES)
    set(out "${CMAKE_BINARY_DIR}/${asset}")
    add_custom_command(
        OUTPUT "${out}"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_SOURCE_DIR}/${asset}" "${out}"
        DEPENDS "${CMAKE_SOURCE_DIR}/${asset}"
        COMMENT "Copying ${asset}"
    )
    list(APPEND ASSET_OUTPUTS "${out}")
endforeach()

add_custom_target(copy_assets DEPENDS ${ASSET_OUTPUTS})
add_dependencies(RayMarching copy_assets)
